CREATE DATABASE IF NOT EXISTS lipson_db;
USE lipson_db;

-- 1. Tabela Única de Usuários
-- Centralizamos aqui todos os funcionários, inclusive os do RH
CREATE TABLE IF NOT EXISTS usuario (
    cpf VARCHAR(11) PRIMARY KEY NOT NULL,
    nome VARCHAR(100) NOT NULL,
    endereco VARCHAR(100) NOT NULL,
    data_admissao DATE NOT NULL,
    data_nascimento DATE,
    cargo VARCHAR(50) NOT NULL, -- Aqui definiremos se é 'RH' ou 'COLABORADOR'
    senha VARCHAR(100) NOT NULL  -- Aumentado para suportar hashes de segurança
);

-- 2. Tabela de Permissões de RH (Opcional, mas recomendada para controle fino)
-- Se o CPF estiver nesta tabela, ele tem acesso às funções de administrador
CREATE TABLE IF NOT EXISTS rh_permissao (
    cpf VARCHAR(11) PRIMARY KEY NOT NULL,
    nivel_acesso INT DEFAULT 1, -- 1 para RH Comum, 2 para Gerente RH
    CONSTRAINT fk_usuario_rh FOREIGN KEY (cpf) REFERENCES usuario(cpf) ON DELETE CASCADE
);
